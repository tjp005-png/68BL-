        new TomSelect("#profile_number", {

            labelField: 'text',
            searchField: 'text',
            create: false, 
            maxOptions: 50,
            load: function(query, callback) {
                if (!query.length) return callback();
                fetch('/api/search_profiles?q=' + encodeURIComponent(query.trim()))
                    .then(response => response.json())
                    .then(json => { callback(json); })
                    .catch(()=>{ callback(); });
            },
            onChange: function(value) {
                if(!value) return;
                fetch(`/api/get_profile_details/${encodeURIComponent(value.trim())}`)
                    .then(res => res.json())
                    .then(data => {
                        if(!data.error) {
                            document.getElementById('generator').value = data.generator || '';
                            document.getElementById('routing_code').value = data.win_code || '';
                            document.getElementById('special_notes').value = data.special_handling || '';
                            
                            let voc = data.voc_percentage;
                            if (voc === 0 || voc === '0.0') voc = 0;
                            document.getElementById('voc_level').value = voc;
                        }
                    });
            }
        });

        const tbody = document.getElementById('sortable-schedule');
        if(tbody) {
            new Sortable(tbody, {

                handle: '.drag-handle', 
                ghostClass: 'sortable-ghost',

                    const rows = tbody.querySelectorAll('tr');
                    let orderData = [];

                        const seriesId = row.getAttribute('data-series');

                                id: id, 
                                order: index,
                                series_id: seriesId

                        }
                    });

                        method: 'POST',

                        body: JSON.stringify({ order: orderData })
                    }).catch(err => console.log('Error saving order:', err));
                }
            });
        }

            rows.forEach(row => {

                let dateVal = this.querySelector('[name="schedule_date"]').value;
                let profileVal = this.querySelector('[name="profile_number"]').value;
                let repeatVal = this.querySelector('[name="repeat_weeks"]') ? this.querySelector('[name="repeat_weeks"]').value : "1";
